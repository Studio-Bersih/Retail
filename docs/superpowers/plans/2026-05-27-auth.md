# Auth / Login Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the SvelteKit project foundation — login page, auth session store with btoa LocalStorage persistence, client-side route guard for `/outlet/**`, and mock user data — so all subsequent feature plans have a working session layer.

**Architecture:** The login page auto-redirects authenticated users (session already in LocalStorage) to `/outlet/dashboard` on mount. On submit, credentials are checked against `mock/auth.ts`; success writes to `$auth` store and navigates. The outlet layout guard (`onMount` check) bounces unauthenticated requests to `/`. `forceWipe()` in `cleaner.ts` is the single logout handler.

**Tech Stack:** SvelteKit 2 (latest), Svelte 5 with runes, TypeScript, TailwindCSS 3 + DaisyUI 4 (night theme), svelte-sonner, Vitest

### Task 3: Auth store

**Files:**
- Create: `src/library/stores/auth.ts`

Create this before the mock — `AuthSession` is defined here and imported by `mock/auth.ts`.

- [ ] **Step 1: Create `src/library/stores/auth.ts`**

```typescript
import { writable } from 'svelte/store'

export type AuthSession = {
    userId: string
    userName: string
    role: 'cashier' | 'manager' | 'admin'
    outletId: string
}

const STORAGE_KEY = 'sb_auth'

function loadFromStorage(): AuthSession | null {
    if (typeof localStorage === 'undefined') return null
    try {
        const raw = localStorage.getItem(STORAGE_KEY)
        if (!raw) return null
        return JSON.parse(atob(raw)) as AuthSession
    } catch {
        return null
    }
}

export const auth = writable<AuthSession | null>(loadFromStorage())

export function setAuth(data: AuthSession): void {
    localStorage.setItem(STORAGE_KEY, btoa(JSON.stringify(data)))
    auth.set(data)
}

export function clearAuth(): void {
    localStorage.removeItem(STORAGE_KEY)
    auth.set(null)
}
```

```svelte
<script lang="ts">
    import { onMount } from 'svelte'
    import { goto } from '$app/navigation'
    import { get } from 'svelte/store'
    import { auth, setAuth } from '$library/stores/auth'
    import { login } from '$library/mock/auth'
    import { toast } from '$library/stores/toast'

    let username = $state('')
    let password = $state('')
    let loading = $state(false)
    let passwordEl = $state<HTMLInputElement | null>(null)

    onMount(() => {
        if (get(auth)) goto('/outlet/dashboard')
    })

    function submit() {
        if (!username.trim() || !password.trim()) return
        loading = true
        try {
            const session = login(username.trim(), password)
            setAuth(session)
            goto('/outlet/dashboard')
        } catch (e: unknown) {
            loading = false
            const msg = e instanceof Error ? e.message : 'Login gagal'
            toast.error(msg === 'useNotice.connection.unauthorized' ? 'Username atau password salah.' : msg)
            password = ''
            passwordEl?.focus()
        }
    }

    function handleKeydown(e: KeyboardEvent) {
        if (e.key === 'Enter') submit()
    }
</script>

<div class="min-h-screen bg-base-300 flex items-center justify-center">
    <div class="card bg-base-100 shadow-xl w-80">
        <div class="card-body items-center text-center gap-4">
            <div>
                <h1 class="text-xl font-bold text-primary">Studio Bersih</h1>
                <p class="text-xs text-base-content/50">Point of Sale</p>
            </div>

            <div class="form-control w-full gap-2">
                <input
                    type="text"
                    class="input input-bordered w-full"
                    placeholder="Username"
                    bind:value={username}
                    onkeydown={handleKeydown}
                    autocomplete="username"
                />
                <input
                    type="password"
                    class="input input-bordered w-full"
                    placeholder="Password"
                    bind:value={password}
                    bind:this={passwordEl}
                    onkeydown={handleKeydown}
                    autocomplete="current-password"
                />
            </div>

            <button
                class="btn btn-primary w-full"
                onclick={submit}
                disabled={loading}
            >
                {#if loading}
                    <span class="loading loading-spinner loading-sm"></span>
                {:else}
                    Masuk
                {/if}
            </button>
        </div>
    </div>
</div>
```
