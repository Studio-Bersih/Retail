# Retail Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the main Retail POS page — Retail/Order mode, member + point redemption, voucher application with cart effects, cart with stock badges, free products section, additional cost/cut, and multi-method payment with Piutang support.

**Architecture:** A single `retail.ts` Svelte store holds all page state (cart, free items, member, voucher, payments, mode, order fields) with derived stores for subtotal, total, and piutangAmount. A pure-function `voucherEngine.ts` handles applying voucher cart effects (testable without Svelte). The page splits across 3 Svelte files: `+page.svelte` orchestrates the left column and inline modals; `CartSection.svelte` renders cart + free products; `PaymentModal.svelte` handles checkout.

**Tech Stack:** SvelteKit · TypeScript · TailwindCSS · DaisyUI · Svelte Stores · Vitest

> **`$lib`** resolves to `src/library/`. Ensure `svelte.config.js` has `kit: { alias: { $lib: 'src/library' } }`.
>
> **Prerequisites:** Working SvelteKit project with TailwindCSS + DaisyUI installed. `src/library/stores/auth.ts` must export a writable `auth` store with shape `{ userId: string; outletId: string; userName: string; role: string }`. `src/library/hooks/useGet.ts` and `usePost.ts` must exist. `src/library/utils/formatter.ts` must export `formatter.currency(n: number): string` using `Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR' })`. If any prerequisite is missing, create it following patterns from prior plans (Akuntansi / Item Transfer).

```typescript
// src/library/stores/retail.ts
import { writable, derived, get } from "svelte/store"
import type {
    Member, RetailCartItem, FreeCartItem, PaymentMethod,
    Voucher, OrderMeta, AdditionalCosts, AdditionalCut
} from "$lib/types/Retail"

// ── State ────────────────────────────────────────────────────────────────────

export const retailMode      = writable<"retail" | "order">("retail")
export const cart            = writable<RetailCartItem[]>([])
export const freeItems       = writable<FreeCartItem[]>([])
export const selectedMember  = writable<Member | null>(null)
export const pointsRedeemed  = writable<number>(0)
export const activeVoucher   = writable<Voucher | null>(null)
// Snapshot of cart/freeItems before the active voucher was applied
export const preVoucherSnap  = writable<{ cart: RetailCartItem[]; freeItems: FreeCartItem[] } | null>(null)

export const additionalCosts = writable<AdditionalCosts>({ packaging: 0, modification: 0, transport: 0, other: 0 })
export const additionalCut   = writable<AdditionalCut>({ fixedAmount: 0, percentage: 0 })
export const notes           = writable<string>("")
export const orderMeta       = writable<OrderMeta | null>(null)
export const payments        = writable<PaymentMethod[]>([{ type: "cash", amount: 0 }])
export const isPiutang       = writable<boolean>(false)
export const transactionType = writable<string>("Private Event")

// ── Derived ──────────────────────────────────────────────────────────────────

export const subtotal = derived(cart, ($cart) =>
    $cart.reduce((sum, item) => sum + item.price * item.qty, 0)
)

export const pointsDiscount = derived(pointsRedeemed, ($pts) => $pts * 50000)

export const cutAmount = derived(
    [subtotal, pointsDiscount, additionalCut],
    ([$sub, $pts, $cut]) => {
        const afterFixed = Math.max(0, $sub - $pts - $cut.fixedAmount)
        const percentCut = Math.floor(afterFixed * ($cut.percentage / 100))
        return $cut.fixedAmount + percentCut
    }
)

export const costsTotal = derived(additionalCosts, ($c) =>
    $c.packaging + $c.modification + $c.transport + $c.other
)

export const total = derived(
    [subtotal, pointsDiscount, cutAmount, costsTotal],
    ([$sub, $pts, $cut, $costs]) => Math.max(0, $sub - $pts - $cut + $costs)
)

export const paymentsTotal = derived(payments, ($pmts) =>
    $pmts.reduce((sum, p) => sum + p.amount, 0)
)

export const piutangAmount = derived(
    [total, paymentsTotal],
    ([$total, $paid]) => Math.max(0, $total - $paid)
)

export const canConfirmPayment = derived(
    [total, paymentsTotal, isPiutang],
    ([$total, $paid, $piutang]) => $piutang || $paid >= $total
)

// ── Cart Actions ─────────────────────────────────────────────────────────────

export function addToCart(item: Omit<RetailCartItem, "isFree"> & { qty?: number }) {
    const qty = item.qty ?? 1
    cart.update(($cart) => {
        const existing = $cart.find((i) => i.id === item.id)
        if (existing) return $cart.map((i) => i.id === item.id ? { ...i, qty: i.qty + qty } : i)
        return [...$cart, { ...item, qty, isFree: false as const }]
    })
}

export function addToFree(item: Omit<FreeCartItem, "isFree"> & { qty?: number }) {
    const qty = item.qty ?? 1
    freeItems.update(($free) => {
        const existing = $free.find((i) => i.id === item.id)
        if (existing) return $free.map((i) => i.id === item.id ? { ...i, qty: i.qty + qty } : i)
        return [...$free, { ...item, qty, isFree: true as const }]
    })
}

export function removeFromCart(itemId: string) {
    cart.update(($cart) => $cart.filter((i) => i.id !== itemId))
}

export function removeFromFree(itemId: string) {
    freeItems.update(($free) => $free.filter((i) => i.id !== itemId))
}

export function updateCartQty(itemId: string, qty: number) {
    if (qty <= 0) { removeFromCart(itemId); return }
    cart.update(($cart) => $cart.map((i) => i.id === itemId ? { ...i, qty } : i))
}

export function updateFreeQty(itemId: string, qty: number) {
    if (qty <= 0) { removeFromFree(itemId); return }
    freeItems.update(($free) => $free.map((i) => i.id === itemId ? { ...i, qty } : i))
}

// ── Payment Actions ──────────────────────────────────────────────────────────

export function addEmoneyPayment() {
    payments.update(($pmts) => [...$pmts, { type: "emoney", provider: "BCA", amount: 0 }])
}

export function removePayment(index: number) {
    payments.update(($pmts) => $pmts.filter((_, i) => i !== index))
}

export function updatePaymentAmount(index: number, amount: number) {
    payments.update(($pmts) => $pmts.map((p, i) => i === index ? { ...p, amount } : p))
}

export function updateEmoneyProvider(index: number, provider: string) {
    payments.update(($pmts) =>
        $pmts.map((p, i) => (i === index && p.type === "emoney" ? { ...p, provider } : p))
    )
}

// ── Voucher Actions ───────────────────────────────────────────────────────────

export function saveVoucherSnapshot() {
    preVoucherSnap.set({ cart: get(cart), freeItems: get(freeItems) })
}

export function restoreVoucherSnapshot() {
    const snap = get(preVoucherSnap)
    if (snap) { cart.set(snap.cart); freeItems.set(snap.freeItems) }
    preVoucherSnap.set(null)
    activeVoucher.set(null)
}

// ── Reset ────────────────────────────────────────────────────────────────────

export function resetRetail() {
    cart.set([])
    freeItems.set([])
    selectedMember.set(null)
    pointsRedeemed.set(0)
    activeVoucher.set(null)
    preVoucherSnap.set(null)
    additionalCosts.set({ packaging: 0, modification: 0, transport: 0, other: 0 })
    additionalCut.set({ fixedAmount: 0, percentage: 0 })
    notes.set("")
    orderMeta.set(null)
```

```typescript
// src/library/utils/voucherEngine.ts
import type { RetailCartItem, FreeCartItem, Voucher, VoucherEffect } from "$lib/types/Retail"
import type { Item } from "$lib/types/Master"

interface CartState {
    cart: RetailCartItem[]
    freeItems: FreeCartItem[]
}

function applyEffect(state: CartState, effect: VoucherEffect, allItems: Item[]): CartState {
    const { cart, freeItems } = state

    if (effect.type === "add_item") {
        const source = allItems.find((i) => i.id === effect.itemId)
        if (!source) return state
        if (effect.toFreeSection) {
            const existing = freeItems.find((i) => i.id === effect.itemId)
            return {
                cart,
                freeItems: existing
                    ? freeItems.map((i) => i.id === effect.itemId ? { ...i, qty: i.qty + effect.qty } : i)
                    : [...freeItems, { id: source.id, name: source.name, sku: source.sku, barcode: source.barcode, qty: effect.qty, stock: source.stock, isFree: true as const }]
            }
        } else {
            const existing = cart.find((i) => i.id === effect.itemId)
            return {
                freeItems,
                cart: existing
                    ? cart.map((i) => i.id === effect.itemId ? { ...i, qty: i.qty + effect.qty } : i)
                    : [...cart, { id: source.id, name: source.name, sku: source.sku, barcode: source.barcode, price: source.price, qty: effect.qty, stock: source.stock, isFree: false as const }]
            }
        }
    }

    if (effect.type === "remove_item") {
        return {
            cart: cart.filter((i) => i.id !== effect.itemId),
            freeItems: freeItems.filter((i) => i.id !== effect.itemId)
        }
    }

    if (effect.type === "set_qty") {
        return {
            cart: cart.map((i) => i.id === effect.itemId ? { ...i, qty: effect.qty } : i),
            freeItems: freeItems.map((i) => i.id === effect.itemId ? { ...i, qty: effect.qty } : i)
        }
    }

    return state
}

export function applyVoucherEffects(
    cart: RetailCartItem[],
    freeItems: FreeCartItem[],
    voucher: Voucher,
    allItems: Item[]
): CartState {
    return voucher.cartEffects.reduce(
        (state, effect) => applyEffect(state, effect, allItems),
        { cart, freeItems }
    )
}
```

```typescript
    const exact = mockItems.find((i) => i.sku.toLowerCase() === q || (i.barcode && i.barcode === q))

        searchInput = ""

        return
    }

    // Partial name / SKU search → open suggestion modal
    searchResults = mockItems.filter(
        (i) => i.name.toLowerCase().includes(q) || i.sku.toLowerCase().includes(q)
    ).slice(0, 8)
    searchHighlight = 0
    searchDest = "cart"
    qtyPromptActive = false
    qtyInput = ""
    showSearchModal = searchResults.length > 0

function handleSearchKeydown(e: KeyboardEvent) {
    if (!showSearchModal) return
    if (e.key === "ArrowDown")  { e.preventDefault(); searchHighlight = Math.min(searchHighlight + 1, searchResults.length - 1) }
    if (e.key === "ArrowUp")    { e.preventDefault(); searchHighlight = Math.max(searchHighlight - 1, 0) }
    if (e.key === "ArrowRight") { e.preventDefault(); searchDest = "free" }
    if (e.key === "ArrowLeft")  { e.preventDefault(); searchDest = "cart" }
    if (e.key === "Enter") {
        e.preventDefault()
        if (qtyPromptActive) {
            commitSearchInsert()
        } else {
            qtyPromptActive = true
            qtyInput = "1"
            setTimeout(() => qtyRef?.focus(), 30)
        }
    }
    if (e.key === "Escape") { closeSearchModal(); e.stopPropagation() }
}

function commitSearchInsert() {
    const qty = parseInt(qtyInput) || 1
    const item = searchResults[searchHighlight]
    if (!item) return
    if (searchDest === "cart") {
        addToCart({ ...item, qty })
    } else {
        addToFree({ id: item.id, name: item.name, sku: item.sku, barcode: item.barcode, stock: item.stock, qty })
    }
    closeSearchModal()
}

    showSearchModal = false
    searchResults = []
    qtyPromptActive = false

    searchInput = ""
```

```typescript
function handleGlobalKeydown(e: KeyboardEvent) {
    // Ctrl+Enter → open payment modal (from anywhere, no modal open)
    if (e.ctrlKey && e.key === "Enter") {
        e.preventDefault()
        if (!showPaymentModal && !showCouponModal && !showSearchModal) {
            showPaymentModal = true
        }
        return
    }
    // ESC priority: close modals first, then focus search
    if (e.key === "Escape") {
        if (showPaymentModal)  { showPaymentModal = false;  return }
        if (showCouponModal)   { showCouponModal = false;   return }
        if (showSearchModal)   { closeSearchModal();        return }
        // No modal open: focus search
        focusSearch()
        return
    }
}
```
