# Pesanan Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the Pesanan (pending order queue) feature — a list of orders from Retail Order mode that cashiers can freely edit until checkout, with admin-gated cancellation.

**Architecture:** Four files: a types module, a mock store with a resolved-cancellation audit log, the main page (table + four inline modals), and an admin-only cancellation review page. Active orders are fully editable in-place without any approval. Checkout calls `logStockMovement` from master-items. Cancellation requires admin approval via a separate queue.

**Tech Stack:** SvelteKit 1.x + Svelte 4, TypeScript 5, TailwindCSS 3, DaisyUI, client-side mocks

import type { KuponCartMutation } from './Kupon'

export interface PesananSnapshot {
    id: string
    outletId: string
    createdBy: string
    memberId: string | null
    items: Array<{
        id: string
        name: string
        sku: string
        barcode: string
        price: number
        qty: number
        stock: number
        isFree: boolean
    }>
    freeItems: Array<{
        id: string
        name: string
        sku: string
        barcode: string
        qty: number
        stock: number
        isFree: true
    }>
    additionalCosts: {
        packaging: number
        modification: number
        transport: number
        other: number
    }
    kupon: { kode: string; nilaiPotongan: number; cartMutations: KuponCartMutation[]; authNip: string | null } | null
    transactionType: string
    notes: string
    orderMeta: {
        orderDate: string        // "YYYY-MM-DD"
        whatsapp: string
        branchId: string
        hour: string             // "HH:MM"
        deliveryType: 'pickup' | 'delivery'
    }
}

export interface PesananVersion {
    index: number
    type: 'original' | 'edit'
    snapshot: PesananSnapshot
    changedFields: string[]
    createdBy: string
    createdAt: string
}

export interface PesananCancellationRequest {
    id: string
    pesananId: string
    status: 'pending'
    reason: string
    requestedBy: string
    requestedAt: string
}

export interface PesananPayment {
    id: string
    type: string
    amount: number
    paidAt: string
}

export type PesananStatus = 'active' | 'awaiting_cancellation' | 'completed' | 'cancelled'

export interface Pesanan {
    id: string
    status: PesananStatus
    currentVersionIndex: number
    versions: PesananVersion[]
    payments: PesananPayment[]
    amountPaid: number
    totalAmount: number
    cancellationRequest: PesananCancellationRequest | null
    createdAt: string
    completedAt: string | null
    cancelledAt: string | null
}

export interface ResolvedCancellationEntry {
    pesananId: string
    outletId: string
    reason: string
    requestedBy: string
    requestedAt: string
    resolvedBy: string
    resolvedAt: string
    resolution: 'approved' | 'rejected'
    rejectionReason: string | null
}

    const itemTotal = snap.items.reduce((s, i) => s + i.price * i.qty, 0)

export function resetPesananStore(): void {
    store = makeSeedStore()

    counter = 4

export function getPesananList(outletId?: string): Pesanan[] {
    const visible = store.filter(p => p.status === 'active' || p.status === 'awaiting_cancellation')
    if (!outletId) return visible
    return visible.filter(p => p.versions[0].snapshot.outletId === outletId)
}

export function getPesananById(id: string): Pesanan | undefined {
    return store.find(p => p.id === id)
}

export function getPendingCancellationRequests(): PesananCancellationRequest[] {
    return store
        .filter(p => p.cancellationRequest?.status === 'pending')
        .map(p => p.cancellationRequest!)
}

export function getResolvedCancellationLog(): ResolvedCancellationEntry[] {
    return resolvedLog
}

export function createPesanan(snapshot: PesananSnapshot): Pesanan {
    counter++
    const id = `PSN-${String(counter).padStart(5, '0')}`
    const version: PesananVersion = {
        index: 0, type: 'original',
        snapshot: { ...snapshot, id },
        changedFields: [], createdBy: snapshot.createdBy, createdAt: new Date().toISOString(),
    }
    const pesanan: Pesanan = {
        id, status: 'active', currentVersionIndex: 0,
        versions: [version], payments: [], amountPaid: 0,
        totalAmount: computeTotal(snapshot),
        cancellationRequest: null, createdAt: new Date().toISOString(), completedAt: null, cancelledAt: null,
    }
    store.push(pesanan)
    return pesanan
}

export function addPayment(id: string, type: string, amount: number): void {
    const pesanan = store.find(p => p.id === id)
    if (!pesanan || pesanan.status !== 'active') return
    pesanan.payments.push({ id: `pay-${Date.now()}`, type, amount, paidAt: new Date().toISOString() })
    pesanan.amountPaid = pesanan.payments.reduce((s, p) => s + p.amount, 0)
}

export function editPesanan(id: string, newSnapshot: PesananSnapshot, userId: string): void {
    const pesanan = store.find(p => p.id === id)
    if (!pesanan || pesanan.status !== 'active') return
    const currentSnap = pesanan.versions[pesanan.currentVersionIndex].snapshot
    const version: PesananVersion = {
        index: pesanan.currentVersionIndex + 1, type: 'edit',
        snapshot: newSnapshot,
        changedFields: getChangedFields(currentSnap, newSnapshot),
        createdBy: userId, createdAt: new Date().toISOString(),
    }
    pesanan.versions.push(version)
    pesanan.currentVersionIndex = version.index
    pesanan.totalAmount = computeTotal(newSnapshot)
}

export function checkoutPesanan(id: string, newPayments: PesananPayment[], userId: string): void {
    const pesanan = store.find(p => p.id === id)
    if (!pesanan || pesanan.status !== 'active') return
    pesanan.payments.push(...newPayments)
    pesanan.amountPaid = pesanan.payments.reduce((s, p) => s + p.amount, 0)
    pesanan.status = 'completed'
    pesanan.completedAt = new Date().toISOString()
}

export function requestCancellation(id: string, userId: string, reason: string): void {
    const pesanan = store.find(p => p.id === id)
    if (!pesanan || pesanan.status !== 'active') return
    pesanan.cancellationRequest = {
        id: `cr-${Date.now()}`, pesananId: id, status: 'pending',
        reason, requestedBy: userId, requestedAt: new Date().toISOString(),
    }
    pesanan.status = 'awaiting_cancellation'
}

export function approveCancellation(id: string, adminId: string): void {
    const pesanan = store.find(p => p.id === id)
    if (!pesanan || pesanan.status !== 'awaiting_cancellation' || !pesanan.cancellationRequest) return
    resolvedLog.push({
        pesananId: id,
        outletId: pesanan.versions[0].snapshot.outletId,
        reason: pesanan.cancellationRequest.reason,
        requestedBy: pesanan.cancellationRequest.requestedBy,
        requestedAt: pesanan.cancellationRequest.requestedAt,
        resolvedBy: adminId, resolvedAt: new Date().toISOString(),
        resolution: 'approved', rejectionReason: null,
    })
    pesanan.cancellationRequest = null
    pesanan.status = 'cancelled'
    pesanan.cancelledAt = new Date().toISOString()
}

export function rejectCancellation(id: string, reason: string, adminId: string): void {
    const pesanan = store.find(p => p.id === id)
    if (!pesanan || pesanan.status !== 'awaiting_cancellation' || !pesanan.cancellationRequest) return
    resolvedLog.push({
        pesananId: id,
        outletId: pesanan.versions[0].snapshot.outletId,
        reason: pesanan.cancellationRequest.reason,
        requestedBy: pesanan.cancellationRequest.requestedBy,
        requestedAt: pesanan.cancellationRequest.requestedAt,
        resolvedBy: adminId, resolvedAt: new Date().toISOString(),
        resolution: 'rejected', rejectionReason: reason,
    })
    pesanan.cancellationRequest = null
    pesanan.status = 'active'
}

<script lang="ts">
    import { get } from 'svelte/store'
    import { goto } from '$app/navigation'
    import { auth } from '$lib/stores/auth'
    import {
        getPendingCancellationRequests,
        getResolvedCancellationLog,
        approveCancellation,
        rejectCancellation,
    } from '$lib/mock/pesanan'
    import type { PesananCancellationRequest, ResolvedCancellationEntry } from '$lib/types/Pesanan'

    const session = get(auth)
    if (session.role !== 'admin') goto('/outlet/pesanan/')

    type Tab = 'menunggu' | 'selesai'
    let activeTab: Tab = 'menunggu'

    let pendingRequests: PesananCancellationRequest[] = getPendingCancellationRequests()
    let resolvedLog: ResolvedCancellationEntry[] = getResolvedCancellationLog()

    function refresh() {
        pendingRequests = getPendingCancellationRequests()
        resolvedLog = getResolvedCancellationLog()
    }

    function handleApprove(pesananId: string) {
        approveCancellation(pesananId, session.userId)
        refresh()
    }

    let rejectOpen = false
    let rejectTargetId = ''
    let rejectReason = ''

    function openReject(pesananId: string) {
        rejectTargetId = pesananId
        rejectReason = ''
        rejectOpen = true
    }
    function submitReject() {
        if (!rejectReason.trim()) return
        rejectCancellation(rejectTargetId, rejectReason.trim(), session.userId)
        rejectOpen = false
        refresh()
    }

    function formatDate(s: string) {
        return new Date(s).toLocaleString('id-ID', {
            day: '2-digit', month: '2-digit', year: 'numeric',
            hour: '2-digit', minute: '2-digit',
        })
    }
</script>
