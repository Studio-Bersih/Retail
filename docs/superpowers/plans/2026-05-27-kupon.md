# Kupon Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement a unified coupon engine merging Coupon Engine v1.0 and Coupon Adaptations v1.0, integrated into Retail, Pesanan, and Perbaikan Transaksi, with a factory admin dashboard and full mock data.

**Architecture:** Pure TypeScript eligibility engine in `utils/couponEligibility.ts` (testable, no side effects). I/O and cart mutations managed by pure functions in `hooks/useCoupon.ts`. Svelte 5 rune-based state in each POS page. Mock data in `mock/kupon.ts` simulates API responses and usage tracking.

**Tech Stack:** SvelteKit 2, Svelte 5 (runes), TypeScript, TailwindCSS + DaisyUI, Vitest (unit tests for eligibility engine and useCoupon utilities)

**Prerequisite:** The SvelteKit project must be initialized before running this plan. If not done, run:
```bash
npx sv create . --template minimal
npm install tailwindcss daisyui svelte-sonner @tailwindcss/vite
```
Then configure TailwindCSS + DaisyUI per SvelteKit docs.

// src/library/types/Kupon.ts
import type { CartItem } from './Cart'

export type KuponKategori = 'Public' | 'Member-only' | 'Personal' | 'Staff/Internal'
export type KuponStatus = 'Active' | 'Inactive'
export type KuponCodeType = 'Standard' | 'Batch' | 'PersonalAuto'
export type KuponMutationType = 'AddItem' | 'RemoveItem' | 'ModifyQty'

export interface KuponCartMutation {
    type: KuponMutationType
    itemId: string
    qty: number
    isFree: boolean       // AddItem only — marks item as free
    priceOverride: number // AddItem only — 0 = use item's default price
}

export interface KuponEffects {
    fixedDiscount: number        // Rp amount off; 0 = not used
    percentageDiscount: number   // % off total; 0 = not used
    cartMutations: KuponCartMutation[]
}

export interface Kupon {
    kode: string
    nama: string
    kategori: KuponKategori
    kodeMember: string | null    // Personal coupons only
    outlet: string[] | null      // null = all outlets
    status: KuponStatus
    tanggalMulai: string         // ISO date string e.g. "2026-01-01"
    tanggalBerakhir: string | null
    minTransaksi: number         // 0 = no minimum
    kuotaTotal: number           // 0 = unlimited
    kuotaPerMember: number       // 0 = unlimited
    butuhOtorisasi: boolean
    syaratKetentuan: string | null
    codeType: KuponCodeType
    effects: KuponEffects
    createdAt: string
    updatedAt: string | null
}

export interface KuponCodePool {
    id: string
    kuponKode: string            // FK → Kupon.kode
    code: string                 // format: {PREFIX6}-{RANDOM8}
    kodeMember: string | null    // PersonalAuto only; null for Batch
    usedAt: string | null
    transactionId: string | null
}

export interface KuponLog {
    id: string
    kodeKupon: string
    idTransaksi: string | null   // null for AuthFailed entries
    kodeMember: string | null
    nipKasir: string
    nipOtorisasi: string | null
    nilaiPotongan: number
    cartMutations: KuponCartMutation[]
    totalSebelum: number
    totalSesudah: number
    outlet: string
    logType: 'Applied' | 'AuthFailed'
    timestamp: string
}

export interface AppliedKupon {
    kode: string
    nama: string
    effects: KuponEffects
    butuhOtorisasi: boolean
    authNip: string | null
    snapshot: Kupon              // cached for local re-validation; no API call needed
    cachedUsage: {
        totalUses: number
        memberUses: number
    }
    cartSnapshot: CartItem[]     // pre-mutation cart; used for full reversal on remove
}

export interface KuponSearchResult {
    kupon: Kupon
    eligibility: EligibilityResult
}

export type EligibilityReason =
    | 'INACTIVE'
    | 'NOT_STARTED'
    | 'EXPIRED'
    | 'QUOTA_EXHAUSTED'
    | 'OUTLET_RESTRICTED'
    | 'MEMBER_REQUIRED'
    | 'MEMBER_MISMATCH'
    | 'PER_MEMBER_QUOTA_EXHAUSTED'
    | 'MIN_TRANSAKSI_NOT_MET'
    | 'AUTHORIZATION_REQUIRED'

export interface EligibilityResult {
    eligible: boolean
    reason?: EligibilityReason
    delta?: number  // only when MIN_TRANSAKSI_NOT_MET
}

export interface EvaluationContext {
    cartTotal: number
    memberKode: string | null
    outletId: string
    timestamp: Date
    usageHistory: {
        totalUses: number
        memberUses: number
    }
    authNip?: string | null  // provided during apply-with-auth flow
}

```typescript
export function resolveKupon(
    enteredCode: string,
    memberKode: string | null
): { kupon: Kupon; poolRow?: KuponCodePool } | { error: string } {
    // Pool lookup first (Batch / PersonalAuto)
    const poolRow = mockKuponCodePool.find(p => p.code === enteredCode)
    if (poolRow) {
        if (poolRow.usedAt !== null) return { error: 'Kode kupon sudah digunakan' }
        if (poolRow.kodeMember !== null && poolRow.kodeMember !== memberKode) {
            return { error: 'Kode kupon ini bukan milik member yang dipilih' }
        }
        const kupon = mockKupon.find(k => k.kode === poolRow.kuponKode)
        if (!kupon) return { error: 'Kupon induk tidak ditemukan' }
        return { kupon, poolRow }
    }
    // Direct lookup (Standard)
    const kupon = mockKupon.find(k => k.kode === enteredCode)
    if (!kupon) return { error: 'Kode kupon tidak ditemukan' }
    return { kupon }
}
```

type Evaluator = (kupon: Kupon, ctx: EvaluationContext) => EligibilityResult

function StatusEvaluator(kupon: Kupon, _ctx: EvaluationContext): EligibilityResult {
    if (kupon.status !== 'Active') return { eligible: false, reason: 'INACTIVE' }
    return { eligible: true }
}

function DateRangeEvaluator(kupon: Kupon, ctx: EvaluationContext): EligibilityResult {
    const now = ctx.timestamp
    const start = new Date(kupon.tanggalMulai)
    if (now < start) return { eligible: false, reason: 'NOT_STARTED' }
    if (kupon.tanggalBerakhir !== null) {
        const end = new Date(kupon.tanggalBerakhir)
        end.setHours(23, 59, 59, 999)
        if (now > end) return { eligible: false, reason: 'EXPIRED' }
    }
    return { eligible: true }
}

// src/library/hooks/useCoupon.ts
import type { Kupon, KuponSearchResult, EligibilityResult, EvaluationContext, AppliedKupon, KuponCartMutation } from '../types/Kupon'
import type { CartItem } from '../types/Cart'
import { evaluate, buildSuggestions } from '../utils/couponEligibility'
import { mockKupon, resolveKupon, getUsageHistory } from '../mock/kupon'

type PartialCtx = { cartTotal: number; memberKode: string | null; outletId: string }

function makeCtx(partial: PartialCtx, usageHistory: { totalUses: number; memberUses: number }, authNip?: string | null): EvaluationContext {
    return { ...partial, timestamp: new Date(), usageHistory, authNip }
}

function applyMutations(cart: CartItem[], mutations: KuponCartMutation[]): CartItem[] {
    const result = cart.map(i => ({ ...i }))
    for (const m of mutations) {
        if (m.type === 'AddItem') {
            const existing = result.find(i => i.id === m.itemId && i.isFree === m.isFree)
            if (existing) { existing.qty += m.qty }
            else { result.push({ id: m.itemId, qty: m.qty, price: m.priceOverride, isFree: m.isFree }) }
        } else if (m.type === 'RemoveItem') {
            const idx = result.findIndex(i => i.id === m.itemId)
            if (idx >= 0) {
                if (result[idx].qty <= m.qty) result.splice(idx, 1)
                else result[idx].qty -= m.qty
            }
        } else if (m.type === 'ModifyQty') {
            const idx = result.findIndex(i => i.id === m.itemId)
            if (idx >= 0) {
                result[idx].qty += m.qty
                if (result[idx].qty <= 0) result.splice(idx, 1)
            }
        }
    }
    return result
}

export async function searchKupon(
    query: string,
    partial: PartialCtx,
    filterEligible: boolean
): Promise<KuponSearchResult[]> {
    const q = query.toLowerCase().trim()
    const candidates = mockKupon.filter(k =>
        k.kode.toLowerCase().includes(q) || k.nama.toLowerCase().includes(q)
    )
    const results: KuponSearchResult[] = candidates.map(kupon => {
        const usageHistory = getUsageHistory(kupon.kode, partial.memberKode)
        const eligibility = evaluate(kupon, makeCtx(partial, usageHistory))
        return { kupon, eligibility }
    })
    if (!filterEligible) return results
    const eligible = results.filter(r => r.eligibility.eligible)
    const suggestions = buildSuggestions(results).filter(s => !eligible.some(e => e.kupon.kode === s.kupon.kode))
    return [...eligible, ...suggestions]
}

export async function getKuponDetail(kode: string): Promise<Kupon | null> {
    return mockKupon.find(k => k.kode === kode) ?? null
}

export async function validateKupon(
    enteredCode: string,
    partial: PartialCtx,
    authNip?: string | null
): Promise<EligibilityResult> {
    const resolved = resolveKupon(enteredCode, partial.memberKode)
    if ('error' in resolved) return { eligible: false, reason: 'INACTIVE' }
    const usageHistory = getUsageHistory(resolved.kupon.kode, partial.memberKode)
    return evaluate(resolved.kupon, makeCtx(partial, usageHistory, authNip))
}

export function buildAppliedKupon(
    kupon: Kupon,
    currentCartItems: CartItem[],
    authNip: string | null,
    memberKode: string | null
): { applied: AppliedKupon; newCartItems: CartItem[] } {
    const cartSnapshot = currentCartItems.map(i => ({ ...i }))
    const newCartItems = applyMutations(currentCartItems, kupon.effects.cartMutations)
    const applied: AppliedKupon = {
        kode: kupon.kode,
        nama: kupon.nama,
        effects: kupon.effects,
        butuhOtorisasi: kupon.butuhOtorisasi,
        authNip,
        snapshot: kupon,
        cachedUsage: getUsageHistory(kupon.kode, memberKode),
        cartSnapshot
    }
    return { applied, newCartItems }
}

export function revertAppliedKupon(applied: AppliedKupon): CartItem[] {
    return applied.cartSnapshot.map(i => ({ ...i }))
}

export function revalidateKupon(applied: AppliedKupon, partial: PartialCtx): EligibilityResult {
    return evaluate(applied.snapshot, makeCtx(partial, applied.cachedUsage, applied.authNip))
}
