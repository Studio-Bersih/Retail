# Konversi Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the Konversi feature — a bidirectional stock conversion system where admins define formulas (e.g., 1 Slop Yakult ↔ 5 Pcs Yakult) and all roles execute them, with full logging and pre-adjustment stock awareness.

**Architecture:** `ConversionFormula` defines a head item and a component list; bidirectional execution (`"buka"` or `"buat"`) is chosen at runtime. `executeConversion()` applies immediate stock changes to `mockItems` and appends a `ConversionLog`. Stock display reuses `getDisplayStock()` from `mock/pre-adjustments.ts`. Formulas are never deleted — only deactivated — to preserve log integrity.

**Tech Stack:** SvelteKit · TypeScript · TailwindCSS · DaisyUI · Vitest

> **`$lib`** resolves to `src/library/`. Ensure `svelte.config.js` has `kit: { alias: { $lib: 'src/library' } }`.
>
> **Prerequisites:** `src/library/stores/auth.ts` exports a writable `auth` store with shape `{ userId: string; outletId: string; userName: string; role: string }`. `src/library/mock/items.ts` exports a mutable `mockItems: { id: string; name: string; stock: number; [key: string]: any }[]`. `mock/pre-adjustments.ts` exports `getDisplayStock(itemId: string, outletId: string): number`.

```typescript
    qty: number

    id: string
    name: string
    headItemId: string
    headQty: number
    components: ConversionComponent[]
    isActive: boolean
    createdBy: string
    createdAt: string
    updatedBy: string | null
    updatedAt: string | null
}

export interface ConversionLog {
    id: string
    formulaId: string
    formulaSnapshot: ConversionFormula
    direction: "buka" | "buat"
    multiplier: number
    outletId: string
    executedBy: string
    executedAt: string
    note: string | null
    usedPreAdjustment: boolean
}

export interface CreateFormulaPayload {
    name: string
    headItemId: string
    headQty: number
    components: ConversionComponent[]
}

export interface UpdateFormulaPayload {
    name?: string
    headItemId?: string
    headQty?: number
    components?: ConversionComponent[]
}

export interface ExecuteConversionPayload {
    formulaId: string
    direction: "buka" | "buat"
    multiplier: number
    outletId: string
    note: string | null
}

export interface FormulaStockSummaryRow {
    itemId: string
    name: string
    role: "head" | "component"
    componentQty: number | null   // null for head row
    realStock: number
    preAdjDelta: number
    displayStock: number          // realStock + preAdjDelta
}
```

```typescript
export function getFormulaStockSummary(formulaId: string, outletId: string): FormulaStockSummaryRow[] {
    const formula = mockFormulas.find(f => f.id === formulaId)
    if (!formula) return []

    const headRealStock = getRealStock(formula.headItemId)
    const headPreAdj = getDisplayStock(formula.headItemId, outletId) - headRealStock

    const rows: FormulaStockSummaryRow[] = [
        {
            itemId: formula.headItemId,
            name: getItemName(formula.headItemId),
            role: "head",
            componentQty: null,
            realStock: headRealStock,
            preAdjDelta: headPreAdj,
            displayStock: headRealStock + headPreAdj
        },
        ...formula.components.map(c => {
            const realStock = getRealStock(c.itemId)
            const preAdjDelta = getDisplayStock(c.itemId, outletId) - realStock
            return {
                itemId: c.itemId,
                name: getItemName(c.itemId),
                role: "component" as const,
                componentQty: c.qty,
                realStock,
                preAdjDelta,
                displayStock: realStock + preAdjDelta
            }
        })
    ]

    return rows
}

export function getConversionLogs(outletId?: string): ConversionLog[] {
    if (!outletId) return [...mockLogs]
    return mockLogs.filter(l => l.outletId === outletId)
}
```

```typescript
export function createConversionFormula(payload: CreateFormulaPayload, adminId: string): ConversionFormula {
    const formula: ConversionFormula = {
        id: `formula-${++idCounter}`,
        name: payload.name,
        headItemId: payload.headItemId,
        headQty: payload.headQty,
        components: payload.components.map(c => ({ ...c })),
        isActive: true,
        createdBy: adminId,
        createdAt: new Date().toISOString(),
        updatedBy: null,
        updatedAt: null
    }
    mockFormulas.push(formula)
    return { ...formula }
}

export function updateConversionFormula(id: string, payload: UpdateFormulaPayload, adminId: string): ConversionFormula {
    const idx = mockFormulas.findIndex(f => f.id === id)
    if (idx === -1) throw new Error(`Formula ${id} not found`)
    mockFormulas[idx] = {
        ...mockFormulas[idx],
        ...(payload.name !== undefined && { name: payload.name }),
        ...(payload.headItemId !== undefined && { headItemId: payload.headItemId }),
        ...(payload.headQty !== undefined && { headQty: payload.headQty }),
        ...(payload.components !== undefined && { components: payload.components.map(c => ({ ...c })) }),
        updatedBy: adminId,
        updatedAt: new Date().toISOString()
    }
    return { ...mockFormulas[idx] }
}

export function deactivateConversionFormula(id: string, adminId: string): void {
    const idx = mockFormulas.findIndex(f => f.id === id)
    if (idx === -1) throw new Error(`Formula ${id} not found`)
    mockFormulas[idx] = { ...mockFormulas[idx], isActive: false, updatedBy: adminId, updatedAt: new Date().toISOString() }
}

export function executeConversion(payload: ExecuteConversionPayload, userId: string): ConversionLog {
    const formula = mockFormulas.find(f => f.id === payload.formulaId)
    if (!formula) throw new Error(`Formula ${payload.formulaId} not found`)

    const sourceItemIds = payload.direction === "buka"
        ? [formula.headItemId]
        : formula.components.map(c => c.itemId)

    const usedPreAdjustment = sourceItemIds.some(itemId => {
        const realStock = mockItems.find(i => i.id === itemId)?.stock ?? 0
        return realStock === 0 && getDisplayStock(itemId, payload.outletId) > 0
    })

    if (payload.direction === "buka") {
        const head = mockItems.find(i => i.id === formula.headItemId)
        if (head) head.stock -= formula.headQty * payload.multiplier
        formula.components.forEach(c => {
            const item = mockItems.find(i => i.id === c.itemId)
            if (item) item.stock += c.qty * payload.multiplier
        })
    } else {
        formula.components.forEach(c => {
            const item = mockItems.find(i => i.id === c.itemId)
            if (item) item.stock -= c.qty * payload.multiplier
        })
        const head = mockItems.find(i => i.id === formula.headItemId)
        if (head) head.stock += formula.headQty * payload.multiplier
    }

    const log: ConversionLog = {
        id: `log-${++idCounter}`,
        formulaId: formula.id,
        formulaSnapshot: { ...formula, components: formula.components.map(c => ({ ...c })) },
        direction: payload.direction,
        multiplier: payload.multiplier,
        outletId: payload.outletId,
        executedBy: userId,
        executedAt: new Date().toISOString(),
        note: payload.note,
        usedPreAdjustment
    }
    mockLogs.push(log)
    return { ...log }
}
```

```svelte
    $: if (search !== undefined || perPage) currentPage = 1
    $: pageButtons = (() => {
        let start = Math.max(1, currentPage - 2)
        let end = Math.min(totalPages, start + 4)
        if (end - start < 4) start = Math.max(1, end - 4)
        return Array.from({ length: end - start + 1 }, (_, i) => start + i)
    })()

    function openCreate() { editingFormula = null; formulaModalOpen = true }
    function openEdit(f: ConversionFormula) { editingFormula = f; formulaModalOpen = true }
    function openKonversi(f: ConversionFormula) { selectedFormula = f; konversiModalOpen = true }

    function handleFormulaSuccess() { allFormulas = getAllConversionFormulas() }
    function handleKonversiSuccess() { if (activeTab === "riwayat") logs = getConversionLogs() }
</script>

<div class="p-6">
    <div class="flex items-center justify-between mb-4">
        <h1 class="text-2xl font-bold">Konversi</h1>
        {#if activeTab === "formula"}
            <button class="btn btn-primary btn-sm" on:click={openCreate}>+ Buat Formula</button>
        {/if}
    </div>

    <div class="tabs tabs-bordered mb-4">
        <button class="tab {activeTab === 'formula' ? 'tab-active' : ''}" on:click={() => activeTab = "formula"}>Formula</button>
        <button class="tab {activeTab === 'riwayat' ? 'tab-active' : ''}" on:click={() => { activeTab = "riwayat"; logs = getConversionLogs() }}>Riwayat Konversi</button>
    </div>

    {#if activeTab === "formula"}
        <div class="flex items-center justify-between gap-4 mb-4">
            <input type="text" class="input input-bordered input-sm w-72" placeholder="Cari formula..." bind:value={formulaSearch} />
            <select class="select select-bordered select-sm" bind:value={formulaPerPage}>
                <option value={10}>10 / halaman</option>
                <option value={25}>25 / halaman</option>
                <option value={50}>50 / halaman</option>
                <option value={100}>100 / halaman</option>
            </select>
        </div>
        <table class="table table-sm w-full">
            <thead>
                <tr><th>Formula</th><th>Produk Utama</th><th>Status</th><th>Aksi</th></tr>
            </thead>
            <tbody>
                {#each paginatedFormulas as f}
                    <tr class="{!f.isActive ? 'opacity-50' : ''}">
                        <td class="font-medium">{f.name}</td>
                        <td class="text-sm text-base-content/70">{f.headItemId}</td>
                        <td>
                            <span class="badge badge-sm {f.isActive ? 'badge-success' : 'badge-ghost'}">
                                {f.isActive ? "Aktif" : "Nonaktif"}
                            </span>
                        </td>
                        <td>
                            <div class="flex gap-2">
                                <button class="btn btn-xs btn-outline" disabled={!f.isActive} on:click={() => openKonversi(f)}>Gunakan</button>
                                <button class="btn btn-xs btn-ghost" on:click={() => openEdit(f)}>Edit</button>
                            </div>
                        </td>
                    </tr>
                {/each}
                {#if paginatedFormulas.length === 0}
                    <tr><td colspan="4" class="text-center text-base-content/40 py-6">Belum ada formula.</td></tr>
                {/if}
            </tbody>
        </table>
        {#if formulaTotalPages > 1}
            <div class="flex justify-center items-center gap-1 mt-4">
                <button class="btn btn-sm btn-ghost" disabled={formulaPage === 1} on:click={() => formulaPage--}>‹</button>
                {#each formulaPageButtons as p}
                    <button class="btn btn-sm {p === formulaPage ? 'btn-primary' : 'btn-ghost'}" on:click={() => formulaPage = p}>{p}</button>
                {/each}
                <button class="btn btn-sm btn-ghost" disabled={formulaPage === formulaTotalPages} on:click={() => formulaPage++}>›</button>
            </div>
        {/if}

    {:else}
        <div class="flex items-center justify-between gap-4 mb-4">
            <input type="text" class="input input-bordered input-sm w-72" placeholder="Cari..." bind:value={search} />
            <select class="select select-bordered select-sm" bind:value={perPage}>
                <option value={10}>10 / halaman</option>
                <option value={25}>25 / halaman</option>
                <option value={50}>50 / halaman</option>
                <option value={100}>100 / halaman</option>
            </select>
        </div>
        <table class="table table-sm w-full">
            <thead>
                <tr><th>Tanggal</th><th>Outlet</th><th>Formula</th><th>Arah</th><th>Jumlah</th><th>Oleh</th><th>Catatan</th></tr>
            </thead>
            <tbody>
                {#each paginated as log}
                    <tr>
                        <td class="text-sm">{new Date(log.executedAt).toLocaleDateString("id-ID", { day: "2-digit", month: "short", year: "numeric", hour: "2-digit", minute: "2-digit" })}</td>
                        <td class="text-sm">{log.outletId}</td>
                        <td class="font-medium">{log.formulaSnapshot.name}</td>
                        <td>
                            <span class="badge badge-sm {log.direction === 'buka' ? 'badge-primary' : 'badge-success'}">
                                {log.direction === "buka" ? "🔓 Buka" : "📦 Buat"}
                            </span>
                        </td>
                        <td>{log.multiplier}×</td>
                        <td>{log.executedBy}</td>
                        <td class="text-base-content/60">{log.note ?? "-"}</td>
                    </tr>
                {/each}
                {#if paginated.length === 0}
                    <tr><td colspan="7" class="text-center text-base-content/40 py-6">Belum ada riwayat konversi.</td></tr>
                {/if}
            </tbody>
        </table>
        {#if totalPages > 1}
            <div class="flex justify-center items-center gap-1 mt-4">
                <button class="btn btn-sm btn-ghost" disabled={currentPage === 1} on:click={() => currentPage--}>‹</button>
                {#each pageButtons as p}
                    <button class="btn btn-sm {p === currentPage ? 'btn-primary' : 'btn-ghost'}" on:click={() => currentPage = p}>{p}</button>
                {/each}
                <button class="btn btn-sm btn-ghost" disabled={currentPage === totalPages} on:click={() => currentPage++}>›</button>
            </div>
        {/if}
    {/if}
</div>

<KonversiModal
    open={konversiModalOpen}
    preselectedFormula={selectedFormula}
    onClose={() => { konversiModalOpen = false; selectedFormula = null }}
    onSuccess={handleKonversiSuccess}
/>

<FormulaModal
    open={formulaModalOpen}
    formula={editingFormula}
    onClose={() => formulaModalOpen = false}
    onSuccess={handleFormulaSuccess}
/>
```
